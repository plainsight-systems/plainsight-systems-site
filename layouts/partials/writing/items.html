{{/* Returns every published write-up as a uniform list of dicts, newest first: local papers, registered project feed snapshots, and curated links. Feed entries win over curated copies with the same URL or canonical URL. Each dict: title, url, kind (paper|article|post), kindLabel, venue, date (YYYY-MM-DD string), about, onsite (true for this site's papers). A data entry missing a required field, or with an unknown kind or a malformed date, fails the build. */}} {{ $kinds := dict "paper" "Paper" "article" "Article" "post" "Post" }} {{ $items := slice }} {{ with site.GetPage "/research" }} {{ range .RegularPages }} {{ $items = $items | append (dict "title" .Title "url" .RelPermalink "kind" "paper" "kindLabel" "Paper" "venue" "Plainsight Systems" "date" (.Date.Format "2006-01-02") "about" "" "onsite" true) }} {{ end }} {{ end }} {{/* Only registered projects participate, even if an old snapshot remains. */}} {{ range site.Data.featured_repos }} {{ $project := . }} {{ with .feed }} {{ $source := index site.Data.project_writing . }} {{ if not $source }} {{ errorf "No writing snapshot for %s; run scripts/refresh-writing.py" . }} {{ end }} {{ range $source.items }} {{ if le (time .published) now }} {{ $items = $items | append (dict "title" .title "url" .url "canonical_url" .canonical_url "kind" "article" "kindLabel" "Article" "venue" $project.writing_name "date" .date "about" $project.writing_name "summary" .summary "onsite" false) }} {{ end }} {{ end }} {{ end }} {{ end }} {{ range site.Data.writing }} {{ $label := index $kinds (.kind | default "") }} {{ if or (not $label) (not .title) (not .url) (not .venue) (not .date) }} {{ errorf "data/writing.yaml: every entry needs title, kind (paper|article|post), url, venue and date; check %q" (.title | default .url) }} {{ end }} {{ $date := printf "%v" .date }} {{ if not (findRE `^\d{4}-\d{2}-\d{2}` $date) }} {{ errorf "data/writing.yaml: date for %q must be YYYY-MM-DD, got %q" .title $date }} {{ end }} {{ $items = $items | append (dict "title" .title "url" .url "kind" .kind "kindLabel" $label "venue" .venue "date" (substr $date 0 10) "about" (.about | default "") "canonical_url" (.canonical_url | default .url) "summary" (.summary | default "") "onsite" false) }} {{ end }} {{ $unique := slice }} {{ $seen := newScratch }} {{ range $items }} {{ $url := strings.TrimSuffix "/" .url }} {{ $canonical := strings.TrimSuffix "/" (.canonical_url | default .url) }} {{ if and (not ($seen.Get $url)) (not ($seen.Get $canonical)) }} {{ $unique = $unique | append . }} {{ $seen.Set $url true }} {{ $seen.Set $canonical true }} {{ end }} {{ end }} {{ return sort $unique "date" "desc" }}
